# Calculo con precedencia de operadores y condiciones compuestas
def evaluar_lote(base, limite, factor):
    acumulado = (base + 15) * factor - 4
    contador = 0

    while contador < limite and acumulado > 0:
        acumulado = acumulado + contador * 2
        contador = contador + 1

    if acumulado >= 100 or base == 50:
        print(acumulado)
        return acumulado
    return 0
